async (page) => {
  const results = [];
  for (const route of ["/auth", "/login"]) {
    for (const [width, height] of [[320,480],[320,568],[360,640],[390,844],[430,932],[768,600],[1440,900]]) {
      await page.setViewportSize({width,height});
      await page.goto("http://127.0.0.1:8081" + route);
      await page.waitForLoadState("networkidle");
      for (const state of route === "/auth" ? ["initial", "error", "password-error", "consent-error"] : ["initial", "error"]) {
        if (state === "error") await page.locator('button[type="submit"]').click();
        if (state === "password-error" || state === "consent-error") {
          await page.locator("#email").fill("p6@example.com");
          await page.locator("#password").fill(state === "password-error" ? "abc" : "Example123");
          await page.locator('button[type="submit"]').click();
        }
        const bounds = await page.evaluate(() => {
          const root = document.documentElement;
          const controls = [...document.querySelectorAll('main input, main button[type="submit"], main [role="checkbox"], main [role="alert"]')];
          return {
            width: root.scrollWidth, height: root.scrollHeight,
            clipped: controls.filter(el => {
              const r = el.getBoundingClientRect();
              return r.top < 0 || r.bottom > innerHeight || r.left < 0 || r.right > innerWidth;
            }).map(el => el.id || el.textContent)
          };
        });
        results.push({route,viewport:{width,height},state,...bounds,
          pass: bounds.width <= width && (width >= 768 || (bounds.height <= height && bounds.clipped.length === 0))});
      }
      if (width === 320 || width === 390 || width === 1440)
        await page.screenshot({path:"output/playwright/p6-"+route.slice(1)+"-"+width+"x"+height+".png"});
    }
  }
  await page.setViewportSize({width:390,height:844});
  await page.goto("http://127.0.0.1:8081/");
  await page.waitForLoadState("networkidle");
  const homeScrollable = await page.evaluate(() => document.documentElement.scrollHeight > innerHeight);
  return {results,homeScrollable};
}
