async (page) => {
  const checks = [];
  const check = (condition, name) => {
    if (!condition) throw new Error(name);
    checks.push(name);
  };
  await page.setViewportSize({ width: 390, height: 600 });
  await page.goto("http://127.0.0.1:8081/");
  await page.waitForLoadState("networkidle");
  const menu = page.getByRole("button", { name: "Open navigation", exact: true });
  await menu.focus();
  await page.keyboard.press("Enter");
  await page.getByRole("navigation", { name: "Mobile main" }).waitFor();
  await page.keyboard.press("Escape");
  await page.getByRole("navigation", { name: "Mobile main" }).waitFor({ state: "hidden" });
  check(
    await menu.evaluate((el) => document.activeElement === el),
    "Public menu Escape restores keyboard focus",
  );
  await page.goto("http://127.0.0.1:8081/dashboard");
  await page.waitForLoadState("networkidle");
  const appMenu = page.getByRole("button", { name: "Open app navigation" });
  await appMenu.focus();
  await page.keyboard.press("Enter");
  await page.getByRole("navigation", { name: "Mobile app navigation" }).waitFor();
  await page.keyboard.press("Escape");
  await page
    .getByRole("navigation", { name: "Mobile app navigation" })
    .waitFor({ state: "hidden" });
  check(
    await appMenu.evaluate((el) => document.activeElement === el),
    "App menu Escape restores keyboard focus",
  );

  await page.setViewportSize({ width: 1440, height: 900 });
  await page.emulateMedia({ reducedMotion: "no-preference" });
  await page.goto("http://127.0.0.1:8081/");
  await page.waitForLoadState("networkidle");
  const card = page.getByTestId("home-traversal");
  await card.scrollIntoViewIfNeeded();
  await page.getByRole("button", { name: "Replay traversal" }).click();
  await page.emulateMedia({ reducedMotion: "reduce" });
  await page.waitForFunction(() => document.documentElement.dataset.reducedMotion === "true");
  check(
    await page.getByRole("button", { name: "Play traversal", exact: true }).isDisabled(),
    "Live reduced-motion change disables autoplay",
  );
  const summary = () => card.getByRole("img").getAttribute("aria-label");
  const paused = await summary();
  await page.waitForTimeout(1200);
  check((await summary()) === paused, "Reduced motion cancels the pending traversal timer");
  await page.getByRole("button", { name: "Replay traversal" }).click();
  const before = await card.boundingBox();
  for (let i = 0; i < 9; i++)
    await page.getByRole("button", { name: "Next traversal step" }).click();
  check(
    (await summary()).includes("8 nodes visited. Queue: empty."),
    "Manual traversal finishes with eight visited nodes and empty queue",
  );
  const after = await card.boundingBox();
  check(
    Math.abs(before.height - after.height) <= 1,
    "Homepage traversal card keeps its height through completion",
  );
  await page.emulateMedia({ reducedMotion: "no-preference" });
  await page.getByRole("button", { name: "Replay traversal" }).click();
  await page.getByRole("button", { name: "Pause traversal" }).click();
  const manualPause = await summary();
  await page.locator("footer").scrollIntoViewIfNeeded();
  await card.scrollIntoViewIfNeeded();
  await page.waitForTimeout(1200);
  check(
    (await summary()) === manualPause,
    "Manual pause survives leaving and re-entering the viewport",
  );

  await page.setViewportSize({ width: 320, height: 480 });
  for (const route of ["/login", "/auth", "/forgot-password", "/reset-password", "/verify-email"]) {
    await page.goto("http://127.0.0.1:8081" + route);
    await page.waitForLoadState("networkidle");
    const submit = page.locator('button[type="submit"]');
    await submit.scrollIntoViewIfNeeded();
    const box = await submit.boundingBox();
    check(box.y >= 0 && box.y + box.height <= 480, route + " submit reachable at 320x480");
  }
  await page.goto("http://127.0.0.1:8081/notifications");
  await page.waitForLoadState("networkidle");
  for (const toggle of await page.getByRole("switch").all()) {
    await toggle.scrollIntoViewIfNeeded();
    check(
      await toggle.evaluate((el) => {
        const r = el.getBoundingClientRect();
        return (
          document.elementFromPoint(r.x + r.width / 2, r.y + r.height / 2) === el ||
          el.contains(document.elementFromPoint(r.x + r.width / 2, r.y + r.height / 2))
        );
      }),
      "Notification preference is visible and reachable: " +
        (await toggle.getAttribute("aria-label")),
    );
  }
  await page.emulateMedia({ reducedMotion: "no-preference" });
  return checks;
}
